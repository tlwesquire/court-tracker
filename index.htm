Building a complete, self-contained HTML app with Options A–D integrated so you can push it to GitHub. Including corrected location data, date-aware mileage, day brief/conflicts, logistics themes, and a seed dataset (your full 287-row docket can replace the seed via localStorage/CSV).Here is a complete single-file app you can save and push to GitHub. It includes Options A–D (corrected courts, date-aware IRS mileage, day brief + conflicts, logistics themes).

Save the file

  1. Create a folder, e.g. court-appearance-tracker
  2. Save the code below as index.html
  3. On GitHub: New repository → upload index.html (and optional README.md)

git init

git add index.html

git commit -m "Court Appearance Tracker — logistics themes, IRS rates, day brief"

git branch -M main

git remote add origin https://github.com/YOUR_USER/court-appearance-tracker.git

git push -u origin main

Enable GitHub Pages (Settings → Pages → Deploy from main / root) to run it at https://YOUR_USER.github.io/court-appearance-tracker/.

Note: Your original 287-row docket was truncated in chat, so this ships with a small seed set. Use Import CSV or paste your full DEFAULT_HEARINGS array into the script. Location migration still rewrites old typo keys in localStorage.

function irsRateForDate(dateStr) {

  if (!dateStr) {

    var t = new Date();

    dateStr = t.getFullYear() + '-' + String(t.getMonth()+1).padStart(2,'0') + '-' + String(t.getDate()).padStart(2,'0');

  }

  var ds = String(dateStr).slice(0, 10);

  for (var i = 0; i < IRS_RATES.length; i++) {

    if (ds >= IRS_RATES[i].from && ds <= IRS_RATES[i].to) return IRS_RATES[i].rate;

  }

  return IRS_RATES[IRS_RATES.length - 1].rate;

}

function normalizeLocation(loc) {

  if (!loc) return '';

  var s = String(loc).toUpperCase().replace(/\s+/g, ' ').trim();

  if (s === 'ZOOM' || s.indexOf('ZOOM') >= 0 || s === 'REMOTE') return 'ZOOM';

  if (LOCATION_KEY_MIGRATIONS[s]) return LOCATION_KEY_MIGRATIONS[s];

  if (s.indexOf('TALLEDAGE') >= 0 || s.indexOf('TALLADEGA') >= 0) return 'TALLADEGA COUNTY - SYLACAUGA';

  if (s.indexOf('DEKLAB') >= 0 || (s.indexOf('DEKALB') >= 0 && (s.indexOf('FT') >= 0 || s.indexOf('FORT') >= 0)))

    return 'DEKALB COUNTY - FORT PAYNE';

  if (s.indexOf('CHAMBERS') >= 0 && (s.indexOf('LAYFATTE') >= 0 || s.indexOf('LAFAYETTE') >= 0))

    return 'CHAMBERS COUNTY - LAFAYETTE';

  if (s.indexOf('TALLAPOOSA') >= 0) return 'TALLAPOOSA COUNTY - ALEXANDER CITY';

  return s;

}

function migrateLocationKeys(list) {

  var changed = false;

  (list || []).forEach(function (h) {

    if (!h || !h.location) return;

    var n = normalizeLocation(h.location);

    if (n && n !== h.location) { h.location = n; changed = true; }

  });

  return changed;

}

function milesOneWay(loc) {

  var key = normalizeLocation(loc);

  return DIST[key] != null ? DIST[key] : 0;

}

function milesRoundTrip(loc) { return milesOneWay(loc) * 2; }

function driveMinOneWay(loc) {

  var key = normalizeLocation(loc);

  return DRIVE_MIN[key] != null ? DRIVE_MIN[key] : 0;

}

function mileageCost(loc, dateStr) {

  return milesRoundTrip(loc) * irsRateForDate(dateStr);

}

function recFee(loc, setting, opts) {

  opts = opts || {};

  var isZoom = !loc || /zoom/i.test(loc) || /zoom/i.test(setting || '');

  if (isZoom) return opts.baseZoom != null ? opts.baseZoom : 100;

  var isLead = opts.isLead !== false;

  var base = opts.baseHearing != null ? opts.baseHearing : 150;

  return Math.round(base + (isLead ? mileageCost(loc, opts.date) : 0));

}

function courtTimeMin(loc, setting) {

  if (/zoom/i.test(loc || '') || /zoom/i.test(setting || '')) return 30;

  return 45;

}

function netPerHour(h, opts) {

  opts = opts || {};

  var fee = Number(h.fee) || 0;

  var isLead = opts.isLead !== false;

  var milesCost = isLead ? mileageCost(h.location, h.date) : 0;

  var net = fee - milesCost;

  var drive = isLead ? driveMinOneWay(h.location) * 2 : 0;

  var hours = Math.max((drive + courtTimeMin(h.location, h.setting)) / 60, 0.25);

  return { fee: fee, milesCost: milesCost, net: net, hours: hours, netPerHr: net / hours, rate: irsRateForDate(h.date), milesRT: isLead ? milesRoundTrip(h.location) : 0 };

}

function feeVerdict(h, opts) {

  var recommended = recFee(h.location, h.setting, { date: h.date, isLead: !opts || opts.isLead !== false });

  var fee = Number(h.fee) || 0;

  var ratio = recommended > 0 ? fee / recommended : 1;

  var label = ratio >= 0.95 ? 'Fair' : ratio >= 0.75 ? 'Under' : 'Low';

  if (ratio > 1.15) label = 'Over';

  return { label: label, recommended: recommended, ratio: ratio };

}

var CONFLICT_BUFFER_MIN = 10;

function todayISO() {

  var t = new Date();

  return t.getFullYear() + '-' + String(t.getMonth()+1).padStart(2,'0') + '-' + String(t.getDate()).padStart(2,'0');

}

function timeToMinutes(t) {

  if (t == null || t === '') return null;

  var s = String(t).trim();

  var am = /\bAM\b/i.test(s), pm = /\bPM\b/i.test(s);

  s = s.replace(/\b[AP]M\b/i, '').trim();

  var p = s.split(':'), h = parseInt(p[0], 10), m = parseInt(p[1] || '0', 10);

  if (isNaN(h)) return null;

  if (isNaN(m)) m = 0;

  if (pm && h < 12) h += 12;

  if (am && h === 12) h = 0;

  return h * 60 + m;

}

function minutesToClock(mins) {

  if (mins == null || isNaN(mins)) return '—';

  mins = ((Math.round(mins) % 1440) + 1440) % 1440;

  var h = Math.floor(mins / 60), m = mins % 60, ap = h >= 12 ? 'PM' : 'AM', h12 = h % 12;

  if (h12 === 0) h12 = 12;

  return h12 + ':' + String(m).padStart(2, '0') + ' ' + ap;

}

function hearingsOnDate(list, dateISO) {

  var d = String(dateISO).slice(0, 10);

  return (list || []).filter(function (h) {

    if (!h || !h.date || String(h.date).slice(0, 10) !== d) return false;

    var r = String(h.result || '').toLowerCase();

    return r !== 'cancelled' && r !== 'canceled';

  }).slice().sort(function (a, b) {

    var ta = timeToMinutes(a.time), tb = timeToMinutes(b.time);

    if (ta == null && tb == null) return 0;

    if (ta == null) return 1;

    if (tb == null) return -1;

    return ta - tb;

  });

}

function dayLeadMap(dayList) {

  var seen = {}, lead = {};

  dayList.forEach(function (h) {

    var key = (h.date || '') + '|' + normalizeLocation(h.location || '');

    if (/zoom/i.test(h.location || '') || /zoom/i.test(h.setting || '')) { lead[h.id] = true; return; }

    if (!seen[key]) { seen[key] = true; lead[h.id] = true; }

    else lead[h.id] = false;

  });

  return lead;

}

function detectDayConflicts(dayList) {

  var conflicts = [], items = [];

  dayList.forEach(function (h) {

    var start = timeToMinutes(h.time);

    if (start == null) return;

    var dur = courtTimeMin(h.location, h.setting);

    items.push({ h: h, start: start, end: start + dur, loc: normalizeLocation(h.location || ''),

      isZoom: /zoom/i.test(h.location || '') || /zoom/i.test(h.setting || '') });

  });

  for (var i = 0; i < items.length; i++) {

    for (var j = i + 1; j < items.length; j++) {

      var a = items[i], b = items[j];

      if (a.start < b.end && b.start < a.end) {

        conflicts.push({ type: 'overlap', severity: 'high', message: 'Time overlap: ' + (a.h.hearing_num || a.h.id) + ' (' + minutesToClock(a.start) + ') & ' + (b.h.hearing_num || b.h.id) + ' (' + minutesToClock(b.start) + ')' });

        continue;

      }

      if (a.end <= b.start && !a.isZoom && !b.isZoom) {

        var gap = b.start - a.end;

        var drive = a.loc === b.loc ? 0 : Math.max(driveMinOneWay(a.loc), driveMinOneWay(b.loc));

        var need = drive + CONFLICT_BUFFER_MIN;

        if (drive > 0 && gap < need) {

          conflicts.push({

            type: gap < drive ? 'impossible' : 'tight_drive',

            severity: gap < drive ? 'high' : 'medium',

            message: (gap < drive ? 'Impossible drive' : 'Tight drive') + ': ' + a.loc + ' → ' + b.loc + ' needs ~' + drive + 'm + buffer; gap ' + gap + 'm'

          });

        }

      }

    }

  }

  return { conflicts: conflicts, ok: conflicts.length === 0 };

}

function buildDayBrief(allHearings, dateISO) {

  dateISO = dateISO || todayISO();

  var day = hearingsOnDate(allHearings, dateISO);

  var leads = dayLeadMap(day);

  var conflictInfo = detectDayConflicts(day);

  var courts = {}, feeSum = 0, netSum = 0, under = [], firstInPerson = null;

  day.forEach(function (h) {

    var loc = normalizeLocation(h.location || '');

    if (loc && loc !== 'ZOOM') courts[loc] = true;

    var isLead = leads[h.id] !== false;

    var np = netPerHour(h, { isLead: isLead });

    feeSum += Number(h.fee) || 0;

    netSum += np.net;

    var v = feeVerdict(h, { isLead: isLead });

    if (v.label === 'Under' || v.label === 'Low') under.push({ h: h, verdict: v });

    var start = timeToMinutes(h.time);

    if (start != null && loc !== 'ZOOM') {

      if (!firstInPerson || start < firstInPerson.start) firstInPerson = { start: start, loc: loc };

    }

  });

  var leaveByLabel = '—', leaveBy = null;

  if (firstInPerson) {

    leaveBy = firstInPerson.start - driveMinOneWay(firstInPerson.loc) - CONFLICT_BUFFER_MIN;

    leaveByLabel = minutesToClock(leaveBy);

  }

  return {

    date: dateISO, count: day.length, courts: Object.keys(courts), courtCount: Object.keys(courts).length,

    feeSum: feeSum, netSum: netSum, underpriced: under, leaveBy: leaveBy, leaveByLabel: leaveByLabel,

    hearings: day, leads: leads, conflicts: conflictInfo.conflicts, conflictOk: conflictInfo.ok

  };

}
